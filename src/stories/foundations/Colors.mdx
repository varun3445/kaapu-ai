export function Swatch({ name, hex, label }) {
  return (
    <div style={{ display:'flex', flexDirection:'column', gap:6 }}>
      <div style={{ height:64, borderRadius:12, background:hex, border:'0.75px solid rgba(0,0,0,0.09)', boxShadow:'0 1px 4px rgba(0,0,0,0.06)' }} />
      <p style={{ fontSize:11, fontWeight:700, color:'#1f2937', margin:0 }}>{name}</p>
      <p style={{ fontSize:10, color:'#818cf8', margin:0, fontFamily:'monospace' }}>{hex}</p>
      <p style={{ fontSize:10, color:'#9ca3af', margin:0 }}>{label}</p>
    </div>
  )
}

export function Grid({ items }) {
  return (
    <div style={{ display:'grid', gridTemplateColumns:'repeat(auto-fill, minmax(130px,1fr))', gap:16, marginBottom:40 }}>
      {items.map(s => <Swatch key={s.name} {...s} />)}
    </div>
  )
}

export function Label({ children }) {
  return <p style={{ fontSize:11, fontWeight:800, color:'#9ca3af', textTransform:'uppercase', letterSpacing:'0.1em', marginBottom:12, marginTop:32 }}>{children}</p>
}
